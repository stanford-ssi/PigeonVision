cmake_minimum_required(VERSION 3.20)
project(pigeonvision_capture VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(PV_BUILD_CAPTURE "Build the Linux libcamera capture executable" ON)
option(PV_BUILD_MEDIA_TESTS "Build FFmpeg sink integration test (requires FFmpeg and JSON development packages)" OFF)
include(CTest)
find_package(Threads REQUIRED)
add_library(pv_core INTERFACE)
target_include_directories(pv_core INTERFACE include)
target_link_libraries(pv_core INTERFACE Threads::Threads)
if(BUILD_TESTING)
  add_executable(pv-core-test tests/core_test.cpp)
  target_link_libraries(pv-core-test PRIVATE pv_core)
  target_compile_options(pv-core-test PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
  add_test(NAME pv-core COMMAND pv-core-test)
  add_executable(pv-framos-test tests/framos_test.cpp)
  target_link_libraries(pv-framos-test PRIVATE pv_core)
  target_compile_options(pv-framos-test PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
  add_test(NAME pv-framos COMMAND pv-framos-test)
  add_executable(pv-dma-heap-test tests/dma_heap_test.cpp)
  target_link_libraries(pv-dma-heap-test PRIVATE pv_core)
  target_compile_options(pv-dma-heap-test PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
  add_test(NAME pv-dma-heap COMMAND pv-dma-heap-test)
endif()
if(PV_BUILD_MEDIA_TESTS AND NOT PV_BUILD_CAPTURE)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavcodec>=61 libavformat>=61 libavutil>=59)
  find_package(nlohmann_json 3.10 REQUIRED)
endif()
if(PV_BUILD_CAPTURE)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Capture requires Linux. Set -DPV_BUILD_CAPTURE=OFF for portable tests.")
  endif()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(CAMERA REQUIRED IMPORTED_TARGET libcamera>=0.5.2)
  pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavcodec>=61 libavformat>=61 libavutil>=59)
  find_package(nlohmann_json 3.10 REQUIRED)
  add_executable(pv-capture src/main.cpp src/config.cpp src/log.cpp src/outputs.cpp src/camera.cpp src/dma_heap.cpp)
  target_link_libraries(pv-capture PRIVATE pv_core PkgConfig::CAMERA PkgConfig::AV nlohmann_json::nlohmann_json)
  target_compile_options(pv-capture PRIVATE -Wall -Wextra -Wpedantic)
  install(TARGETS pv-capture RUNTIME DESTINATION bin)
endif()
if(BUILD_TESTING AND (PV_BUILD_CAPTURE OR PV_BUILD_MEDIA_TESTS))
  add_executable(pv-encoder-input-test tests/encoder_input_test.cpp)
  target_link_libraries(pv-encoder-input-test PRIVATE pv_core PkgConfig::AV)
  target_compile_options(pv-encoder-input-test PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
  add_test(NAME pv-encoder-input COMMAND pv-encoder-input-test)
  add_executable(pv-colour-test tests/colour_test.cpp)
  target_link_libraries(pv-colour-test PRIVATE pv_core nlohmann_json::nlohmann_json)
  target_compile_options(pv-colour-test PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
  add_test(NAME pv-colour COMMAND pv-colour-test)
endif()
if(PV_BUILD_MEDIA_TESTS)
  add_executable(pv-media-test tests/media_test.cpp src/outputs.cpp src/config.cpp src/log.cpp)
  target_link_libraries(pv-media-test PRIVATE pv_core PkgConfig::AV nlohmann_json::nlohmann_json)
  target_compile_options(pv-media-test PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_compile_definitions(pv-media-test PRIVATE CLOCK_BOOTTIME=CLOCK_MONOTONIC)
  endif()
  add_executable(pv-config-test tests/config_test.cpp src/config.cpp)
  target_link_libraries(pv-config-test PRIVATE pv_core nlohmann_json::nlohmann_json)
  target_compile_options(pv-config-test PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_compile_definitions(pv-config-test PRIVATE CLOCK_BOOTTIME=CLOCK_MONOTONIC)
  endif()
  add_test(NAME pv-config COMMAND pv-config-test)
endif()
